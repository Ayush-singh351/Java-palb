class Solution {

    public int findMinDiff(int arr[], int m) {

        java.util.Arrays.sort(arr);

        int min = Integer.MAX_VALUE;

        for(int i=0;i+m-1<arr.length;i++){

            int diff = arr[i+m-1] - arr[i];

            if(diff < min)
                min = diff;
        }

        return min;
    }
}
